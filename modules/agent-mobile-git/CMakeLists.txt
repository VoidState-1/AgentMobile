cmake_minimum_required(VERSION 3.22)
project(AgentMobileGit LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(USE_SSH OFF CACHE STRING "" FORCE)
set(USE_GSSAPI OFF CACHE BOOL "" FORCE)
set(USE_SHA1 CollisionDetection CACHE STRING "" FORCE)
set(USE_SHA256 Builtin CACHE STRING "" FORCE)
set(USE_HTTP_PARSER builtin CACHE STRING "" FORCE)
set(REGEX_BACKEND builtin CACHE STRING "" FORCE)
if(ANDROID)
  # Only app-owned POSIX directories are used; opt out of the upstream FAT-era no-op.
  add_compile_definitions(AGENT_MOBILE_POSIX_PERMISSIONS=1)
  # TLS is registered via Android SSLSocket; no private CA store or bundled OpenSSL.
  set(USE_HTTPS OFF CACHE STRING "" FORCE)
elseif(APPLE)
  set(USE_HTTPS SecureTransport CACHE STRING "" FORCE)
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/libgit2/include/git2.h")
  message(FATAL_ERROR "Run npm run prepare:git before building native Git")
endif()
add_subdirectory(vendor/libgit2 EXCLUDE_FROM_ALL)
add_library(agent_mobile_git_core STATIC cpp/probe.cpp cpp/repository.cpp)
target_include_directories(agent_mobile_git_core PUBLIC cpp vendor/libgit2/include PRIVATE vendor/json/include)
target_link_libraries(agent_mobile_git_core PUBLIC libgit2package)
if(ANDROID)
  add_library(agent_mobile_git SHARED android/src/main/cpp/bridge.cpp)
  target_link_libraries(agent_mobile_git PRIVATE agent_mobile_git_core)
  target_link_options(agent_mobile_git PRIVATE "-Wl,-z,max-page-size=16384")
endif()
